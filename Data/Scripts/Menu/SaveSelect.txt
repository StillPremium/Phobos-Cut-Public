private alias object.value1  : object.fade
private alias object.value2  : object.currentSelection
private alias object.value3  : object.flashTimer
private alias object.value4  : object.spikesPos.x
private alias object.value5  : object.spikesPos.y
private alias object.value6  : object.titlePos.y
private alias object.value7  : object.returning
private alias object.value8  : object.save1Char
private alias object.value9  : object.save2Char
private alias object.value10 : object.save3Char
private alias object.value11 : object.save4Char
private alias object.value12 : object.nSaveChar
private alias object.value13 : object.arrowFlash
private alias object.value14 : object.deleteSel
private alias object.value15 : object.pressedUD
private alias object.value16 : object.pressedLR
private alias object.value17 : object.pressedA
private alias object.value18 : object.deleteSelected

// States
private alias 0 : SAVESEL_INIT
private alias 1 : SAVESEL_SLIDEIN
private alias 2 : SAVESEL_CONTROLS
private alias 3 : SAVESEL_FLASHOPTION
private alias 4 : SAVESEL_CREATESAVE
private alias 5 : SAVESEL_LOADSAVE
private alias 6 : SAVESEL_RETURNING

// Saves
private alias 0 : SAVESEL_NOSAVE
private alias 1 : SAVESEL_SAVE1
private alias 2 : SAVESEL_SAVE2
private alias 3 : SAVESEL_SAVE3
private alias 4 : SAVESEL_SAVE4
private alias 5 : SAVESEL_BACK

// Player List Pos Aliases
public alias 0 : PLAYER_SONIC_A
public alias 1 : PLAYER_TAILS_A
public alias 2 : PLAYER_KNUCKLES_A
public alias 3 : PLAYER_SONIC_TAILS_A
public alias 4 : PLAYER_KNUCKLES_TAILS_A
public alias 5 : PLAYER_AMY_A
public alias 6 : PLAYER_AMY_TAILS_A

// Game Mode Aliases
private alias 0 : MODE_NOSAVE
private alias 1 : MODE_SAVEGAME
private alias 2 : MODE_TIMEATTACK


// ========================
// Events
// ========================

event ObjectUpdate
	if object.state != SAVESEL_RETURNING
		CheckTouchRect(0, 0, screen.xsize, screen.ysize)
		if checkResult > -1
			options.physicalControls = false
		end if
	else
		options.physicalControls = true
	end if

	if options.physicalControls == false
		temp0 = screen.xcenter
		temp0 -= 56
		CheckTouchRect(0, 96, temp0, screen.ysize)
		if checkResult > -1
			arrayPos0 = checkResult
			temp0 = touchscreen[arrayPos0].xpos
			temp0 -= 60	
			temp1 = touchscreen[arrayPos0].ypos
			temp1 -= 184
			ATan2(temp2, temp0, temp1)
			temp2 += 0x20
			temp2 &= 0xFF
			temp2 >>= 6
			switch temp2
			case 0
				keyDown[0].right = true
				if object.pressedLR == false
					keyPress[0].right |= keyDown[0].right
				end if
				object.pressedLR = keyDown[0].right
				break

			case 1
				keyDown[0].down = true
				if object.pressedUD == false
					keyPress[0].down |= keyDown[0].down
				end if
				object.pressedUD = keyDown[0].down
				break

			case 2
				keyDown[0].left = true
				if object.pressedLR == false
					keyPress[0].left |= keyDown[0].left
				end if
				object.pressedLR = keyDown[0].left
				break

			case 3
				keyDown[0].up = true
				if object.pressedUD == false
				keyPress[0].up |= keyDown[0].up
				end if
				object.pressedUD = keyDown[0].up
				break
				
			end switch
		else
			object.pressedLR = false
			object.pressedUD = false
		end if

		temp0 = screen.xsize
		temp0 -= 128
		CheckTouchRect(temp0, 96, screen.xsize, screen.ysize)
		if checkResult > -1
			keyDown[0].buttonA = true
			if object.pressedA == false
				keyPress[0].buttonA |= keyDown[0].buttonA
			end if
			object.pressedA = keyDown[0].buttonA
		else
			object.pressedA = false
		end if
	end if

	switch object.state
	case SAVESEL_INIT
		object.spikesPos.x = -34
		object.titlePos.y = -26
		object.currentSelection = SAVESEL_SAVE1
		stage.playerListPos = PLAYER_SONIC_A
		stage.player2Enabled = false

		if screen.xsize > 320
			EditFrame(1, -91, 0, 182, 26, 99, 1)		// "SAVE SELECT"
		else
			EditFrame(1, -61, 0, 136, 26, 105, 173)		// "SAVE SEL"
		end if
		object.state++
		if stage.lastMenu == 1
			stage.lastMenu = 0

			object.spikesPos.x = 0
			object.titlePos.y = 5

			object.state = SAVESEL_FADEIN
			object.fade = 256

			object.currentSelection = options.saveSlot
			object.currentSelection++

			SetScreenFade(0, 0, 0, 255)
		end if
		break

	case SAVESEL_SLIDEIN
		object.spikesPos.x += 4
		if object.spikesPos.x > -4
			object.spikesPos.x = 0
			object.state++
		end if

		if object.alpha < 250
			object.alpha += 8
		else
			object.alpha = 255
		end if

		if object.titlePos.y < 1
			object.titlePos.y += 4
		else
			object.titlePos.y = 5
		end if
		break

	case SAVESEL_CONTROLS
		object.flashTimer++
		object.flashTimer %= 12

		object.arrowFlash++
		object.arrowFlash %= 32

		if options.physicalControls == false
			CheckTouchRect(0, 0, screen.xsize, screen.ysize)
			if checkResult > -1
				temp0 = object.titlePos.y
				temp0 += 26
				
				CheckTouchRect(36, object.titlePos.y, 98, temp0)
				if checkResult > -1
					object.currentSelection = SAVESEL_BACK
				else
					if object.currentSelection == SAVESEL_BACK
						object.currentSelection = SAVESEL_SAVE1
					end if
				end if

				arrayPos1 = object.currentSelection	// Get the save
				arrayPos1--
				arrayPos1 *= 8				// Get the character of that save
				arrayPos1 += 4				// Get the stage of that save
				CheckEqual(saveRAM[arrayPos1], 0)
				temp1 = checkResult
				CheckEqual(object.currentSelection, SAVESEL_NOSAVE)
				temp1 |= checkResult
				if temp1 == false
					temp1 = screen.xsize
					temp1 -= 83
					temp2 = screen.xsize
					temp2 -= 5
					CheckTouchRect(temp1, object.titlePos.y, temp2, temp0)
					if checkResult > -1
						object.deleteSelected = true
					else
						object.deleteSelected = false
					end if
				else
					object.deleteSelected = false
				end if
			else
				options.physicalControls = true

				if object.currentSelection == SAVESEL_BACK
					object.state = SAVESEL_RETURNING
				end if

				if object.deleteSelected == true
					object.state = SAVESEL_DELETING
					object.deleteSelected = false

					PlaySfx(SfxName[Lamp Post], false)
				end if
			end if
		end if

		if object.alpha < 250
			object.alpha += 8
		else
			object.alpha = 255
		end if

		if object.titlePos.y < 1
			object.titlePos.y += 4
		else
			object.titlePos.y = 5
		end if

		if keyPress[0].left == true
			object.currentSelection--
			if object.currentSelection < SAVESEL_NOSAVE
				object.currentSelection = SAVESEL_SAVE4
			end if

			options.physicalControls = true
			object.flashTimer = 0
			object.arrowFlash = 0

			PlaySfx(SfxName[Score Add], false)
		end if

		if keyPress[0].right == true
			object.currentSelection++
			if object.currentSelection > SAVESEL_SAVE4
				object.currentSelection = SAVESEL_NOSAVE
			end if

			options.physicalControls = true
			object.flashTimer = 0
			object.arrowFlash = 0

			PlaySfx(SfxName[Score Add], false)
		end if

		arrayPos1 = object.currentSelection	// Get the save
		arrayPos1--
		arrayPos1 *= 8				// Get the character of that save
		arrayPos1 += 4				// Get the stage of that save
		CheckEqual(saveRAM[arrayPos1], 0)
		temp0 = checkResult
		CheckEqual(object.currentSelection, SAVESEL_NOSAVE)
		temp0 |= checkResult
		if temp0 == true
			temp0 = PLAYER_KNUCKLES_TAILS_A
#platform: USE_ORIGINS
			if game.hasPlusDLC == true
				temp0 = PLAYER_AMY_TAILS_A
			end if
#endplatform

			if keyPress[0].up == true
				switch object.currentSelection
				case SAVESEL_NOSAVE
					object.nSaveChar++
					if object.nSaveChar > temp0
						object.nSaveChar = PLAYER_SONIC_A
					end if
					break

				case SAVESEL_SAVE1
					object.save1Char++
					if object.save1Char > temp0
						object.save1Char = PLAYER_SONIC_A
					end if
					break

				case SAVESEL_SAVE2
					object.save2Char++
					if object.save2Char > temp0
						object.save2Char = PLAYER_SONIC_A
					end if
					break

				case SAVESEL_SAVE3
					object.save3Char++
					if object.save3Char > temp0
						object.save3Char = PLAYER_SONIC_A
					end if
					break

				case SAVESEL_SAVE4
					object.save4Char++
					if object.save4Char > temp0
						object.save4Char = PLAYER_SONIC_A
					end if
					break
					
				end switch

				PlaySfx(SfxName[Score Add], false)
			end if

			if keyPress[0].down == true
				switch object.currentSelection
				case SAVESEL_NOSAVE
					object.nSaveChar--
					if object.nSaveChar < PLAYER_SONIC_A
						object.nSaveChar = temp0
					end if
					break

				case SAVESEL_SAVE1
					object.save1Char--
					if object.save1Char < PLAYER_SONIC_A
						object.save1Char = temp0
					end if
					break

				case SAVESEL_SAVE2
					object.save2Char--
					if object.save2Char < PLAYER_SONIC_A
						object.save2Char = temp0
					end if
					break

				case SAVESEL_SAVE3
					object.save3Char--
					if object.save3Char < PLAYER_SONIC_A
						object.save3Char = temp0
					end if
					break

				case SAVESEL_SAVE4
					object.save4Char--
					if object.save4Char < PLAYER_SONIC_A
						object.save4Char = temp0
					end if
					break
					
				end switch

				PlaySfx(SfxName[Score Add], false)
			end if
		else
			if keyPress[0].buttonX == true
				object.state = SAVESEL_DELETING

				PlaySfx(SfxName[Lamp Post], false)
			end if
		end if

		object.deleteSel = false

		temp0 = keyPress[0].start
		temp0 |= keyPress[0].buttonA
		if temp0 == true
			object.state++
			StopMusic()

			PlaySfx(SfxName[Menu Select], false)
		end if

		if keyPress[0].buttonB == true
			options.physicalControls = true
			object.state = SAVESEL_RETURNING
		end if
		break

	case SAVESEL_FLASHOPTION
		object.flashTimer += 3
		object.flashTimer %= 12

		object.arrowFlash++
		object.arrowFlash %= 32

		if object.alpha < 250
			object.alpha += 8
		else
			object.alpha = 255
		end if

		object.timer++
		if object.timer >= 36
			object.flashTimer = 0
			object.timer = 0
			arrayPos1 = object.currentSelection	// Get the save
			arrayPos1--
			arrayPos1 *= 8				// Get the character of that save
			arrayPos1 += 4				// Get the stage of that save

			CheckEqual(saveRAM[arrayPos1], 0)
			temp0 = checkResult
			CheckEqual(object.currentSelection, SAVESEL_NOSAVE)
			temp0 |= checkResult
			if temp0 == true
				object.state = SAVESEL_CREATESAVE
			else
				object.state = SAVESEL_LOADSAVE
			end if
		end if
		break

	case SAVESEL_CREATESAVE
		object.flashTimer += 3
		object.flashTimer %= 12

		object.arrowFlash++
		object.arrowFlash %= 32

		object.fade += 8

		if object.fade >= 256
			arrayPos1 = object.currentSelection	// Get the save
			arrayPos1--

			arrayPos1 *= 8				// Get the character of that save

			switch object.currentSelection
			case SAVESEL_NOSAVE
				temp0 = object.nSaveChar
				break

			case SAVESEL_SAVE1
				temp0 = object.save1Char
				break

			case SAVESEL_SAVE2
				temp0 = object.save2Char
				break

			case SAVESEL_SAVE3
				temp0 = object.save3Char
				break

			case SAVESEL_SAVE4
				temp0 = object.save4Char
				break
				
			end switch

			saveRAM[arrayPos1] = temp0
			stage.playerListPos = temp0

			arrayPos1++				// Get the lives of that save
			saveRAM[arrayPos1] = 3
			player.lives = 3

			arrayPos1++				// Get the score of that save
			saveRAM[arrayPos1] = 0
			player.score = 0

			arrayPos1++				// Get the score bonus of that save
			saveRAM[arrayPos1] = 50000
			player.scoreBonus = 50000

			arrayPos1++				// Get the stage of that save
			saveRAM[arrayPos1] = 1
			stage.listPos = 0

			arrayPos1++				// Get the emerald count of that save
			saveRAM[arrayPos1] = 0
			specialStage.emeralds = 0

			arrayPos1++				// Get the special stage of that save
			saveRAM[arrayPos1] = 0
			specialStage.listPos = 0

			stage.activeList = REGULAR_STAGE

			options.saveSlot = object.currentSelection
			options.saveSlot--
			if object.currentSelection == SAVESEL_NOSAVE
				options.gameMode = MODE_NOSAVE
			else
				options.gameMode = MODE_SAVEGAME
			end if

			WriteSaveRAM()
			LoadStage()

			SetScreenFade(0, 0, 0, 255)
		end if

		SetScreenFade(0, 0, 0, object.fade)
		break

	case SAVESEL_LOADSAVE
		object.flashTimer += 3
		object.flashTimer %= 12

		object.fade += 8

		if object.fade >= 256
			arrayPos1 = object.currentSelection	// Get the save
			arrayPos1--

			arrayPos1 *= 8				// Get the character of that save
			stage.playerListPos = saveRAM[arrayPos1]

			arrayPos1++				// Get the lives of that save
			player.lives = saveRAM[arrayPos1]

			arrayPos1++				// Get the score of that save
			player.score = saveRAM[arrayPos1]

			arrayPos1++				// Get the score bonus of that save
			player.scoreBonus = saveRAM[arrayPos1]

			arrayPos1++				// Get the stage of that save
			stage.listPos = saveRAM[arrayPos1]
			stage.listPos--
			if stage.listPos < 0
				stage.listPos = 0
				player.lives = 3
				player.score = 0
			end if

			arrayPos1++				// Get the emerald count of that save
			specialStage.emeralds = saveRAM[arrayPos1]

			arrayPos1++				// Get the special stage of that save
			specialStage.listPos = saveRAM[arrayPos1]

			stage.activeList = REGULAR_STAGE

			arrayPos1 = object.currentSelection	// Get the save
			arrayPos1--
			arrayPos1 *= 8				// Get the character of that save
			arrayPos1 += 4				// Get the stage of that save
			if saveRAM[arrayPos1] >= 80 		// Check if the save is on a special stage
				specialStage.nextZone = saveRAM[arrayPos1]
				specialStage.nextZone -= 129
				stage.activeList = SPECIAL_STAGE
#platform: USE_ORIGINS
				stage.activeList = 3
#endplatform

				stage.listPos = specialStage.listPos
			end if

			options.saveSlot = object.currentSelection
			options.saveSlot--
			options.gameMode = MODE_SAVEGAME

			lampPostID = 0

			LoadStage()

			SetScreenFade(0, 0, 0, 255)
		end if

		SetScreenFade(0, 0, 0, object.fade)
		break

	case SAVESEL_RETURNING
		object.flashTimer++
		object.flashTimer %= 12

		object.arrowFlash++
		object.arrowFlash %= 32

		object.alpha -= 24

		object.spikesPos.x += 10
		if object.spikesPos.x >= 80
			if object.returning == false
				ResetObjectEntity(1, TypeName[Main Menu], 0, 0, 0)
				object.returning = true
				object[1].returning = true
			end if
		end if
		break

	case SAVESEL_DELETING
		object.flashTimer++
		object.flashTimer %= 12

		object.arrowFlash++
		object.arrowFlash %= 32

		temp0 = keyPress[0].up
		temp0 |= keyPress[0].down
		temp0 |= keyPress[0].left
		temp0 |= keyPress[0].right
		if temp0 == true
			object.deleteSel ^= true

			PlaySfx(SfxName[Score Add], false)
		end if

		temp0 = keyPress[0].start
		temp0 |= keyPress[0].buttonA
		if temp0 == true
			if object.deleteSel == true
				PlaySfx(SfxName[Lose Rings], false)

				arrayPos1 = object.currentSelection	// Get the save
				arrayPos1--

				arrayPos1 *= 8				// Get the character of that save

				switch object.currentSelection
				case SAVESEL_SAVE1
					object.save1Char = PLAYER_SONIC_A
					break

				case SAVESEL_SAVE2
					object.save2Char = PLAYER_SONIC_A
					break

				case SAVESEL_SAVE3
					object.save3Char = PLAYER_SONIC_A
					break

				case SAVESEL_SAVE4
					object.save4Char = PLAYER_SONIC_A
					break
					
				end switch

				saveRAM[arrayPos1] = PLAYER_SONIC_A

				arrayPos1++				// Get the lives of that save
				saveRAM[arrayPos1] = 3

				arrayPos1++				// Get the score of that save
				saveRAM[arrayPos1] = 0

				arrayPos1++				// Get the score bonus of that save
				saveRAM[arrayPos1] = 50000

				arrayPos1++				// Get the stage of that save
				saveRAM[arrayPos1] = 0

				arrayPos1++				// Get the emerald count of that save
				saveRAM[arrayPos1] = 0

				arrayPos1++				// Get the special stage of that save
				saveRAM[arrayPos1] = 0

				WriteSaveRAM()
			else
				PlaySfx(SfxName[Lamp Post], false)
			end if

			object.state = SAVESEL_CONTROLS
		end if
		break

	case SAVESEL_FADEIN
		object.flashTimer++
		object.flashTimer %= 12

		if object.fade > 3
			object.fade -= 8
			SetScreenFade(0, 0, 0, object.fade)
		else
			object.fade = 0
			object.state = SAVESEL_CONTROLS
		end if
		break
		
	end switch

	object.rotation += 2
	object.rotation &= 511

	object.spikesPos.y++
	object.spikesPos.y %= 156

	object.rotation = 256
end event


event ObjectDraw
	LoadSpriteSheet("Menu/SaveSelect.gif")

	if object.propertyValue == 0
		DrawRect(0, 0, screen.xsize, screen.ysize, 0xE0, 0xE0, 0x00, 0xFF)

		arrayPos0 = 0
		while arrayPos0 < 255
			temp0 = arrayPos0
			temp0 *= 24
			DrawRect(0, temp0, screen.xsize, screen.ysize, 0xE0, 0xC0, 0x00, arrayPos0)
			arrayPos0++
		loop

		if object.state >= SAVESEL_SLIDEIN
			DrawSpriteScreenXY(1, screen.xcenter, object.titlePos.y)		// "SAVE SELECT"

			DrawSpriteScreenXY(18, 36, object.titlePos.y)				// B button + "BACK"

			if object.currentSelection == SAVESEL_BACK
				if object.flashTimer < 6
					DrawRect(36, object.titlePos.y, 62, 2, 0xF8, 0x00, 0x00, 0xFF)
					DrawRect(36, object.titlePos.y, 2, 26, 0xF8, 0x00, 0x00, 0xFF)

					DrawRect(96, object.titlePos.y, 2, 26, 0xF8, 0x00, 0x00, 0xFF)

					temp0 = object.titlePos.y
					temp0 += 24
					DrawRect(36, temp0, 62, 2, 0xF8, 0x00, 0x00, 0xFF)
				end if
			end if

			arrayPos1 = object.currentSelection	// Get the save
			arrayPos1--
			arrayPos1 *= 8				// Get the character of that save
			arrayPos1 += 4				// Get the stage of that save
			CheckEqual(saveRAM[arrayPos1], 0)
			temp0 = checkResult
			CheckEqual(object.currentSelection, SAVESEL_NOSAVE)
			temp0 |= checkResult
			if temp0 == false
				DrawSpriteScreenXY(69, screen.xsize, object.titlePos.y)			// X button + "DELETE"

				if object.deleteSelected == true
					if object.flashTimer < 6
						temp1 = screen.xsize
						temp1 -= 83
						temp2 = screen.xsize
						temp2 -= 7

						DrawRect(temp1, object.titlePos.y, 78, 2, 0xF8, 0x00, 0x00, 0xFF)
						DrawRect(temp1, object.titlePos.y, 2, 26, 0xF8, 0x00, 0x00, 0xFF)

						DrawRect(temp2, object.titlePos.y, 2, 26, 0xF8, 0x00, 0x00, 0xFF)

						temp0 = object.titlePos.y
						temp0 += 24
						DrawRect(temp1, temp0, 78, 2, 0xF8, 0x00, 0x00, 0xFF)
					end if
				end if
			end if
		end if
	else
		arrayPos0 = 0
		while arrayPos0 < 255
			temp0 = arrayPos0
			temp0 *= 24
			temp0 += screen.yoffset
			DrawRect(0, temp0, screen.xsize, screen.ysize, 0xE0, 0xC0, 0x00, arrayPos0)
			arrayPos0++
		loop
	end if

	temp0 = screen.xcenter
	temp0 -= 167
	temp0 = 45
	temp1 = screen.yoffset
	temp1 += 62
	if screen.xsize <= 384
		temp0 = screen.xcenter
		temp0 -= 34
		temp1 = screen.yoffset
		temp1 += 33
	end if
	DrawSpriteScreenXY(2, temp0, temp1)							// No Save Box

	if object.currentSelection == SAVESEL_NOSAVE
		if object.flashTimer < 6
			DrawSpriteScreenXY(15, temp0, temp1)					// No Save Box Outline
		end if
		DrawSpriteScreenXY(81, temp0, temp1)						// Selected No Save Header
	end if

	temp2 = object.nSaveChar
	temp2 += 8
	temp1 -= 38

	if object.currentSelection == SAVESEL_NOSAVE
		if object.arrowFlash < 16
			temp0 += 33
			temp1 += 30
			DrawSpriteScreenXY(17, temp0, temp1)					// Blue Arrow
			temp1 += 34
			temp1 += 28

			DrawSpriteScreenFX(17, FX_ROTATE, temp0, temp1)				// Blue Arrow
			temp0 -= 33
			temp1 -= 34
			temp1 -= 30
			temp1 -= 28
		end if
	end if

	DrawSpriteScreenXY(temp2, temp0, temp1)							// Character portrait
	temp1 += 38

	if screen.xsize <= 384
		if screen.xsize > 320
			temp0 = 50
		else
			temp0 = 40
		end if
		temp1 = screen.yoffset
		temp1 += 109
	else
		temp3 = screen.xsize
		temp3 /= 96
		temp2 = screen.xsize
		temp2 /= 6
		temp2 += temp3
		temp0 += temp2
	end if

	arrayPos0 = 0
	while arrayPos0 < 4
		DrawSpriteScreenXY(3, temp0, temp1)						// Save Box

		arrayPos0++
		if object.currentSelection == arrayPos0
			if object.deleteSelected == false
				if object.flashTimer < 6
