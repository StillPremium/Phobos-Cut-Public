event ObjectUpdate
	switch object.state
	case ACHIEVEMENTS_INIT
		object.spikesPos.x = screen.xsize
		object.spikesPos.x += 34
		object.titlePos.y = -26
		object.ach1Offset = 362
		object.ach2Offset = 362
		object.ach3Offset = 362
		object.scrollingUp = true
		object.state++
		break

	case ACHIEVEMENTS_SLIDEIN
		object.spikesPos.x -= 4
		temp0 = screen.xsize
		temp0 += 4
		if screen.xsize <= 320
			temp0 += 16
		end if
		if object.spikesPos.x < temp0
			object.spikesPos.x = screen.xsize
			if screen.xsize <= 320
				object.spikesPos.x += 16
			end if
		end if

		object.ach1Offset -= 24
		if object.ach1Offset < 24
			object.ach1Offset = 0
		end if

		if object.ach1Offset <= 240
			object.ach2Offset -= 24
			if object.ach2Offset < 24
				object.ach2Offset = 0
			end if
		end if

		if object.ach2Offset <= 240
			object.ach3Offset -= 24
			if object.ach3Offset < 24
				object.ach3Offset = 0
				object.state++
			end if
		end if

		if object.alpha < 250
			object.alpha += 8
		else
			object.alpha = 255
		end if

		if object.titlePos.y < 1
			object.titlePos.y += 4
		else
			object.titlePos.y = 5
		end if
		break

	case ACHIEVEMENTS_CONTROLS
		if object.alpha < 250
			object.alpha += 8
		else
			object.alpha = 255
		end if

		CheckTouchRect(0, 0, screen.xsize, screen.ysize)
		if checkResult > -1
			options.physicalControls = false
		end if

		if options.physicalControls == false
			CheckTouchRect(0, 0, screen.xsize, screen.ysize)
			if checkResult > -1
				CheckTouchRect(0, 96, screen.xsize, screen.ysize)
				if checkResult > -1
					CheckTouchRect(0, 178, screen.xsize, screen.ysize)
					if checkResult > -1
						keyDown[0].down = true
						if object.pressedUD == false
							keyPress[0].down |= keyDown[0].down
						end if
						object.pressedUD = keyDown[0].down
					else
						keyDown[0].up = true
						if object.pressedUD == false
							keyPress[0].up |= keyDown[0].up
							end if
						object.pressedUD = keyDown[0].up
					end if
					object.pressedLR = false
				else
					temp0 = object.titlePos.y
					temp0 += 26
						
					CheckTouchRect(5, object.titlePos.y, 67, temp0)
					if checkResult > -1
						object.backSelected = true
						object.pressedLR = false
						object.pressedUD = false
					else
						if object.backSelected == true
							object.backSelected = false
						end if
					end if
				end if
			else
				options.physicalControls = true

				if object.backSelected == true
					object.state = ACHIEVEMENTS_RETURNING
				end if
			end if
		else
			object.pressedLR = false
			object.pressedUD = false
		end if

		temp0 = object.currentSelection
		temp0--
		if object.scrollingUp == true
			temp0++
		end if

		if temp0 >= ACHIEVEMENTS_BEATTHECLOCK
			temp0 = ACHIEVEMENTS_BEATTHECLOCK
		end if
		object.scrollAmount = temp0
		object.scrollAmount *= 80
		object.scrollAmount += temp0
		object.scrollAmount += temp0

		object.flashTimer++
		object.flashTimer %= 12

		if object.titlePos.y < 1
			object.titlePos.y += 4
		else
			object.titlePos.y = 5
		end if

		if keyPress[0].up == true
			if object.currentSelection >= ACHIEVEMENTS_RAMPRINGACRO
				options.physicalControls = true
				object.scrollingUp = true

				object.currentSelection--

				object.flashTimer = 0

				PlaySfx(SfxName[Score Add], false)
			end if
		end if

		if keyPress[0].down == true
			if object.currentSelection <= ACHIEVEMENTS_BEATTHECLOCK
				options.physicalControls = true
				object.scrollingUp = false

				object.currentSelection++

				object.flashTimer = 0

				PlaySfx(SfxName[Score Add], false)
			end if
		end if

		if keyPress[0].buttonB == true
			options.physicalControls = true
			object.state = ACHIEVEMENTS_RETURNING
		end if
		break

	case ACHIEVEMENTS_FLASHOPTION
		object.flashTimer += 3
		object.flashTimer %= 12

		object.timer++
		if object.timer >= 36
			object.flashTimer = 0
			object.timer = 0

			object.state = ACHIEVEMENTS_CASEDEPENDANT
		end if
		break

	case ACHIEVEMENTS_CASEDEPENDANT

		break

	case ACHIEVEMENTS_RETURNING
		object.flashTimer++
		object.flashTimer %= 12

		object.alpha -= 24

		object.spikesPos.x -= 8
		temp0 = screen.xsize
		temp0 -= 40
		if object.spikesPos.x <= temp0
			if object.returning == false
				ResetObjectEntity(1, TypeName[Extras], 1, 0, 0)
				object.returning = true
				object[1].currentSelection = 1
				object[1].state = 47
				screen.xoffset = object.spikesPos.x
				screen.xoffset -= screen.xsize

				object[1].flashTimer++
				object[1].flashTimer %= 12
			else
				screen.xoffset = object.spikesPos.x
				screen.xoffset -= screen.xsize

				object[1].flashTimer++
				object[1].flashTimer %= 12
			end if
		end if

		if object.spikesPos.x <= 0
			ResetObjectEntity(0, TypeName[Extras], 0, 0, 0)
			object[0].flashTimer = object[1].flashTimer
			object[0].currentSelection = object[1].currentSelection
			screen.xoffset = 0

			ResetObjectEntity(1, TypeName[Blank Object], 0, 0, 0)
		end if
		break
		
	end switch

	object.rotation += 2
	object.rotation &= 511

	object.spikesPos.y--
	if object.spikesPos.y < 0
		object.spikesPos.y = 156
	end if
end event


event ObjectDraw
	SetActivePalette(2, 0, screen.ysize)
	LoadSpriteSheet("Menu/Achievements.gif")

	if object.propertyValue == 0
		DrawRect(0, 0, screen.xsize, screen.ysize, 0x2D, 0x99, 0x23, 0xFF)
	end if

	temp0 = 49
	if screen.xsize <= 384
		temp0 = 20
		if screen.xsize <= 320
			temp0 = 0
		end if
	end if
	temp1 = camera[0].ypos
	temp1 += 50
	temp1 -= object.scrollAmount

	CallNativeFunction(GetAchievementCount)
	temp3 = checkResult
	arrayPos0 = 0
	while arrayPos0 < temp3
		if arrayPos0 == 0
			temp0 -= object.ach1Offset
		end if
		if arrayPos0 == 1
			temp0 -= object.ach2Offset
		end if
		if arrayPos0 == 2
			temp0 -= object.ach3Offset
		end if
		if object.state == 47
			temp0 -= screen.xsize
		end if
		if object.state >= ACHIEVEMENTS_SLIDEIN
			DrawSpriteScreenXY(3, temp0, temp1)					// Achievement Box

			if object.currentSelection == arrayPos0
				if object.backSelected == false
					if object.flashTimer < 6
						DrawSpriteScreenXY(28, temp0, temp1)		// Achievement Box Outline
					end if
				end if
			end if

			CallNativeFunction2(GetAchievement, arrayPos0, 0)
			if checkResult == false
				LoadSpriteSheet("Menu/AchGreyIcon.gif")
			end if

			temp2 = arrayPos0
			temp2 += 4
			DrawSpriteScreenXY(temp2, temp0, temp1)					// Icon

			CallNativeFunction2(GetAchievement, arrayPos0, 0)
			if checkResult == false
				LoadSpriteSheet("Menu/Achievements.gif")
			end if

			temp2 = arrayPos0
			temp2 += 16
			DrawSpriteScreenXY(temp2, temp0, temp1)					// Name & Description
		end if
		if arrayPos0 == 0
			temp0 += object.ach1Offset
		end if
		if arrayPos0 == 1
			temp0 += object.ach2Offset
		end if
		if arrayPos0 == 2
			temp0 += object.ach3Offset
		end if
		if object.state == 47
			temp0 += screen.xsize
		end if
		temp1 += 82
		arrayPos0++
	loop

	if object.propertyValue == 0
		if object.state >= ACHIEVEMENTS_SLIDEIN
			temp0 = screen.xcenter
			if screen.xsize <= 320
				temp0 += 16
			end if
			DrawSpriteScreenXY(1, temp0, object.titlePos.y)				// "ACHIEVEMENTS"

			DrawSpriteScreenXY(2, 5, object.titlePos.y)				// B button + "BACK"

			if object.backSelected == true
				if object.flashTimer < 6
					DrawRect(5, object.titlePos.y, 62, 2, 0xF8, 0x00, 0x00, 0xFF)
					DrawRect(5, object.titlePos.y, 2, 26, 0xF8, 0x00, 0x00, 0xFF)

					DrawRect(65, object.titlePos.y, 2, 26, 0xF8, 0x00, 0x00, 0xFF)

					temp0 = object.titlePos.y
					temp0 += 24
					DrawRect(5, temp0, 62, 2, 0xF8, 0x00, 0x00, 0xFF)
				end if
			end if

			DrawRect(object.spikesPos.x, 0, screen.xsize, screen.ysize, 0xF8, 0xF8, 0x00, 0xFF)

			arrayPos0 = object.spikesPos.y
			arrayPos0 /= 2
			arrayPos0 -= 78
			temp0 = screen.ysize
			temp0 += 78
			while arrayPos0 <= temp0
				DrawSpriteScreenXY(0, object.spikesPos.x, arrayPos0)		// Spikes
				arrayPos0 += 78
			loop
		end if
	end if

	SetActivePalette(0, 0, screen.ysize)

	object.inkEffect = INK_ALPHA
	if screen.xsize > 384
		temp0 = -2
	else
		if screen.xsize > 320
			temp0 = -19
		else
			temp0 = screen.xsize
			temp0 -= 60
		end if
	end if

	if engine.deviceType == STANDARD
		temp0 = -64
	end if

	LoadSpriteSheet("Menu/Options.gif")
	DrawSpriteScreenFX(29, FX_INK, temp0, 151)

	if keyDown[0].up == true
		DrawSpriteScreenFX(30, FX_INK, temp0, 151)
	end if

	if keyDown[0].down == true
		DrawSpriteScreenFX(31, FX_INK, temp0, 151)
	end if
end event
