	LoadSpriteSheet("Menu/TimeAttack.gif")

	if object.propertyValue == 0
		if object.gradientAlphaOffset > 0
			object.gradientAlphaOffset--
		else
			object.gradientAlphaOffset = 0
		end if

		GetPaletteEntry(0, 201, temp4)
		temp1 = temp4
		temp2 = temp4
		temp3 = temp4

		temp2 &= 0x00FF00
		temp3 &= 0x0000FF
  
		temp1 >>= 16
		temp2 >>= 8

		DrawRect(0, 0, screen.xsize, screen.ysize, temp1, temp2, temp3, 0xFF)

		arrayPos0 = 0
		while arrayPos0 < 255
			temp0 = arrayPos0
			temp0 *= 24
			temp0 += screen.yoffset

			temp1 -= object.gradientAlphaOffset
			DrawRect(0, temp0, screen.xsize, screen.ysize, 0xE0, 0xC0, 0x00, temp1)
			arrayPos0 += 2
		loop
	end if

	CheckLower(object.state, TATTACK_PART2_SLIDEIN)
	temp0 = checkResult
	CheckEqual(object.state, TATTACK_PART2_RETURNING)
	temp0 |= checkResult
	if temp0 == true
		temp0 = screen.xsize
		temp0 -= 257


		temp0 = screen.xcenter
		temp0 -= 111


		temp1 = camera[0].ypos
		temp1 += 50
		temp1 -= object.scrollAmount

		arrayPos0 = 0
		while arrayPos0 < 8
			temp2 = arrayPos0
			temp2 -= object.currentSelection
			Abs(temp2)
			if temp2 == 0
				temp0 += object.button1Offset
			end if
			if temp2 == 1
				temp0 += object.button2Offset
			end if
			if temp2 == 2
				temp0 += object.button3Offset
			end if
			if temp2 == 3
				temp0 += object.button4Offset
			end if
			if object.state == 47
				temp0 += screen.xsize
			end if
			if object.state >= TATTACK_PART1_SLIDEIN
				DrawSpriteScreenXY(3, temp0, temp1)						// Zone Box

				if object.currentSelection == arrayPos0
					if object.backSelected == false
						if object.flashTimer < 6
							DrawSpriteScreenXY(4, temp0, temp1)			// Zone Box Outline
						end if
					end if
				end if

				LoadSpriteSheet("LevelSelect/Icons.gif")
				SetActivePalette(1, 0, screen.ysize)

				temp2 = arrayPos0
				temp2 += 12
				DrawSpriteScreenFX(temp2, FX_SCALE, temp0, temp1)				// Icon

				LoadSpriteSheet("Menu/TimeAttack.gif")
				SetActivePalette(3, 0, screen.ysize)
