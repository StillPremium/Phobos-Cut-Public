event ObjectUpdate
	// Progress HUD flashing
	// 0 ring flashing takes priority over time over flashing
	if player[0].rings == 0
		object.flashTimer++
		if object.flashTimer == 16
			object.flashTimer = 0
		end if
	else
		if stage.minutes == 9
			object.flashTimer++
			if object.flashTimer == 16
				object.flashTimer = 0
			end if
		else
			if options.gameMode == MODE_TIMEATTACK
				if stage.timeEnabled == false
					object.flashTimer++
					if object.flashTimer == 16
						object.flashTimer = 0
					end if
				end if
			end if
		end if
	end if

#platform: USE_ORIGINS
	if game.timeOver != false
		object.timeOver = true
	end if
#endplatform

	if object[SLOT_PAUSEMENU].type == TypeName[Pause Menu]
		if object.priority == PRIORITY_ALWAYS
			object.priority = PRIORITY_ACTIVE
		end if
	else
		if object.priority == PRIORITY_ACTIVE
			object.priority = PRIORITY_ALWAYS
		end if
	end if

	if options.gameMode == MODE_TIMEATTACK
		if stage.timeEnabled == true
			if stage.state == STAGE_RUNNING
				if keyPress[0].buttonY == true
					PlaySfx(SfxName[Teleport], false)
				end if
				if keyDown[0].buttonY == true
					object.alpha += 16
					if object.alpha >= 320
						stage.quickRestart = true
						LoadStage()
					end if
				else
					if object.alpha > 13
						object.alpha -= 32
					else
						object.alpha = 0
					end if
				end if
			end if
		else
			if stage.state == STAGE_RUNNING
				if object.alpha > 32
					if stage.quickRestart == true
						if object.alpha > 256
							if music.currentTrack != 0
								PlayMusic(0)
							end if
						end if
						stage.quickRestart = false
					end if
					object.alpha -= 32
				else
					object.alpha = 0
				end if
			end if
		end if
	end if

	object.inkEffect = INK_ALPHA
	object.superAlpha = 160
	if player[0].gravity == GRAVITY_AIR
		if player[0].animation == ANI_JUMPING
			if player[0].jumpAbilityState == 1
				object.superAlpha = 255
			end if
		end if
	end if

	if options.gameMode != MODE_TIMEATTACK
		CheckGreater(player[0].rings, 49)
		temp0 = checkResult
		CheckEqual(specialStage.emeralds, 0x7F)
		temp0 &= checkResult
		CheckEqual(options.superStates, true)
		temp0 &= checkResult
		if Player_superState == SUPERSTATE_SUPER
			CheckEqual(player[0].animation, ANI_SUPER_TRANSFORM)
			temp0 &= checkResult
		end if
		CheckNotEqual(player[0].controlMode, CONTROLMODE_NONE)
		temp0 &= checkResult
		CheckEqual(stage.timeEnabled, true)
		temp0 &= checkResult
	else
		CheckNotEqual(player[0].controlMode, CONTROLMODE_NONE)
		temp0 = checkResult
		CheckEqual(stage.timeEnabled, true)
		temp0 &= checkResult
	end if
	if temp0 == true
		if player[0].animation == ANI_SUPER_TRANSFORM
			if object.superOffset < 56
				if object.superOffset >= 8
					object.superOffset -= 8
				end if
			end if
		else
			if object.superOffset >= 8
				object.superOffset -= 8
			end if
		end if
	else
		if object.superOffset <= 56
			object.superOffset += 8
		end if
	end if
end event


event ObjectDraw	
#platform: USE_ORIGINS
	if stage.playerListPos == PLAYER_AMY_A
		EditFrame(15, 0, 0, 16, 16, 239, 174)
		EditFrame(21, 0, 0, 31, 7, 182, 183)
	end if
#endplatform

	CallFunction(object.drawFunction)

	temp2 = object.alpha
	object.alpha = object.superAlpha
	if options.gameMode == MODE_TIMEATTACK
		object.alpha = 255
	end if

	checkResult = true

	if object.drawFunction == HUD_Draw_Mobile
		CheckNotEqual(player[0].type, TypeName[Debug Mode])
	end if

	if checkResult == true
		LoadSpriteSheet("Global/Items.gif")

		temp0 = screen.xsize
		temp0 -= 61
		temp0 += object.superOffset
		temp1 = 47
		if object.drawFunction != HUD_Draw_Mobile
			if player[0].type == TypeName[Debug Mode]
				object.alpha = 0
			end if
		end if
		DrawSpriteScreenFX(46, FX_INK, temp0, temp1)

		if options.gameMode == MODE_TIMEATTACK
			DrawSpriteScreenFX(47, FX_INK, temp0, temp1)
		end if
	end if

	object.alpha = temp2

	if options.gameMode == MODE_TIMEATTACK
		if stage.timeEnabled == true
			if object.alpha > 13
				DrawRect(0, 0, screen.xsize, screen.ysize, 224, 224, 224, object.alpha)
			end if
		else
			if object.alpha > 32
				DrawRect(0, 0, screen.xsize, screen.ysize, 224, 224, 224, object.alpha)
			end if
		end if	
	end if

	LoadSpriteSheet("Global/Display.gif")
end event
