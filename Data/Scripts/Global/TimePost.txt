event ObjectUpdate
	object.ypos += object.yvel
	ObjectTileCollision(CSIDE_FLOOR, 0, 28, PATH_A)
	if checkResult == true
		object.yvel = 0
	else
		object.yvel += 0x3800
	end if

	switch object.state
	case TIMEPOST_AWAITPLAYER
		object.drawOrder = 4
		if options.gameMode == MODE_TIMEATTACK
			if object.propertyValue == TIMEPOST_FINISHGATE
				temp0 = screen.xoffset
				temp0 += screen.xsize
				if object.ixpos <= temp0
					stage.newXBoundary1 = object.xpos
					stage.newXBoundary1 >>= 16
					stage.newXBoundary1 -= screen.xcenter
				end if

				stage.newXBoundary2 = object.xpos
				stage.newXBoundary2 >>= 16
				stage.newXBoundary2 += screen.xcenter

				object.priority = PRIORITY_ACTIVE
			else
				object.priority = PRIORITY_ACTIVE
			end if

			object.frame = 0

			temp0 = object.ypos
			temp0 -= player[0].ypos
			temp0 >>= 16
			temp0 += 24
			FlipSign(temp0)
			if player[0].xpos <= object.xpos
				BoxCollisionTest(C_TOUCH, object.entityPos, -12, temp0, 80, 28, 0, -8, -1, 8, 1)
			else
				BoxCollisionTest(C_TOUCH, object.entityPos, -80, temp0, 12, 28, 0, -8, -1, 8, 1)
			end if
			if player[0].speed == 0
				checkResult = false
			end if
			if checkResult == true
				if player[0].xpos <= object.xpos
					object.spinDirection = FACING_RIGHT
				else
					object.spinDirection = FACING_LEFT
				end if
				object.spinCount = 16
				object.state = TIMEPOST_SPINNING
				PlaySfx(SfxName[Sign Post], false)

				if object.propertyValue == TIMEPOST_FINISHGATE
					stage.timeEnabled = false

					player[0].controlMode 	= CONTROLMODE_NONE
					player[0].interaction 	= false
					player[0].up 			= false
					player[0].down 			= false
					player[0].left 			= false
					player[0].right 		= false
					player[0].jumpPress 	= false
					player[0].jumpHold 		= false
				else
					stage.timeEnabled = true
				end if
			end if
		else
			object.type = TypeName[Blank Object]
		end if

		if object.propertyValue == TIMEPOST_FINISHGATE
			temp0 = 40
			temp0 += screen.xcenter
			temp0 <<= 16
			temp0 += object.xpos
			if player[0].xpos > temp0
				player[0].xpos = temp0
				player[0].speed = 0
				player[0].xvel = 0
			end if
		end if
		break

	case TIMEPOST_SPINNING
		object.timer++
		if object.timer == 2
			object.timer = 0
			object.animationTimer++
			if object.animationTimer >= 4
				object.animationTimer = 0
				Rand(temp0, 48)
				temp0 <<= 16
				temp0 += object.xpos
				temp0 -= 0x180000
				Rand(temp1, 18)
				temp1 <<= 16
				temp1 += object.ypos
				temp1 += 0x40000
				CreateTempObject(TypeName[Ring Sparkle], 0, temp0, temp1)
				object[tempObjectPos].drawOrder = 4

				if object.spinCount > 0
					object.spinCount--
				else
					if object.propertyValue == TIMEPOST_FINISHGATE
						object.state = TIMEPOST_BLANK

						PlayMusic(TRACK_ACTFINISH)
						object[SLOT_ACTFINISH].type = TypeName[Act Finish]
						object[SLOT_ACTFINISH].drawOrder = 6
						object[SLOT_ACTFINISH].priority = PRIORITY_ACTIVE
					else
						object.state = TIMEPOST_AWAITPLAYER
					end if
				end if
			end if

			if object.spinDirection == FACING_RIGHT
				object.frame++
			else
				if object.frame == 0
					object.frame = 4
				end if
				object.frame--
			end if
			if object.frame > 3
				object.frame = 0
			end if
		end if

		if object.propertyValue == TIMEPOST_FINISHGATE
			music.volume--

			temp0 = 40
			temp0 += screen.xcenter
			temp0 <<= 16
			temp0 += object.xpos
			if player[0].xpos > temp0
				player[0].xpos = temp0
				player[0].speed = 0
				player[0].xvel = 0
			end if
		end if
		break
		
	// "TIMEPOST_BLANK" doesn't have anything
	
	end switch
end event


event ObjectDraw
	if object.propertyValue == TIMEPOST_STARTGATE
		if object.spinCount > 0
			DrawSprite(object.frame)
		else
			DrawSprite(0)
		end if
	else
		if object.spinCount > 0
			temp0 = object.frame
			temp0 += 4
			DrawSprite(temp0)
		else
			DrawSprite(4)
		end if
	end if
end event


event ObjectStartup
	LoadSpriteSheet("Global/Items.gif")
	
	// Start Gate
	SpriteFrame(-23, -28, 46, 56, 256, 1)
	SpriteFrame(-19, -28, 39, 56, 303, 1)
	SpriteFrame(-22, -28, 44, 56, 384, 1)
	SpriteFrame(-20, -28, 40, 56, 343, 1)
	
	// Finish Gate
	SpriteFrame(-23, -28, 46, 56, 256, 58)
	SpriteFrame(-19, -28, 39, 56, 303, 58)
	SpriteFrame(-22, -28, 44, 56, 384, 58)
	SpriteFrame(-20, -28, 40, 56, 343, 58)
	
	if options.gameMode == MODE_TIMEATTACK
		object[24].type = TypeName[Time Post]
		object[25].priority = PRIORITY_ACTIVE
		object[24].xpos = player[0].xpos
		object[24].ixpos += 96
		object[24].ypos = player[0].ypos
		object[24].yvel = -0x10000
		CheckCurrentStageFolder("Zone04")
		if checkResult == true
			if stage.actNum == 4
				object[24].ixpos -= 136
				object[24].iypos += 136
			end if
		end if
		object[24].spinCount = 16

		foreach(TypeName[Lamp Post], arrayPos0, ALL_ENTITIES)
			object[arrayPos0].type = TypeName[Blank Object]
		next

		foreach(TypeName[Sign Post], arrayPos0, ALL_ENTITIES)
			object[arrayPos0].type = TypeName[Time Post]
			object[arrayPos0].propertyValue = TIMEPOST_FINISHGATE
			object[arrayPos0].yvel = -0x10000
			object[arrayPos0].spinCount = 16
		next
	end if

	SetPaletteEntry(0, 74, 0x202020)
	SetPaletteEntry(1, 74, 0x202020)
end event
